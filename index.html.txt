<script>
  let selectedTransport = 136;
  let selectedHotel = 352;

  const activities = [
    ["09:30","Passeio pelo Bairro Gótico","Explorar as ruas históricas e pequenas praças","Explorar"],
    ["12:30","Almoço no centro","Restaurante local próximo do itinerário","Gastronomia"],
    ["15:00","Sagrada Família","Visita e tempo livre na zona envolvente","Cultura"],
    ["18:30","Passeio pelo Passeig de Gràcia","Arquitetura, lojas e pausa para café","Passeio"],
    ["20:30","Jantar","Sugestão adaptada ao teu ritmo","Gastronomia"]
  ];

  function renderItinerary(){
    const pace = document.getElementById('pace').value;
    const subtitle = document.getElementById('itinerary-subtitle');
    subtitle.textContent = `Gerado a partir das tuas escolhas e do ritmo ${pace.toLowerCase()}.`;
    const container = document.getElementById('itinerary-list');
    const days = [
      ["Dia 1","Chegada e primeiro contacto"],
      ["Dia 2","Centro histórico e cultura"],
      ["Dia 3","Arquitetura e gastronomia"],
      ["Dia 4","Tempo livre e descoberta"]
    ];
    container.innerHTML = days.map((day, index) => {
      let dayActivities = activities;
      if(pace === "Relaxado") dayActivities = activities.filter((_,i)=>[0,1,3].includes(i));
      if(pace === "Intenso") dayActivities = [...activities,["22:00","Noite livre","Sugestão de zona para continuar a explorar","Livre"]];
      if(index === 0) dayActivities = [
        ["10:55","Chegada a Barcelona","Transfer para o alojamento","Transporte"],
        ["13:00","Check-in","Deixar bagagem e instalar","Alojamento"],
        ["16:00","Passeio de adaptação","Primeiro contacto com o bairro","Passeio"],
        ["20:00","Jantar","Sugestão próxima do hotel","Gastronomia"]
      ];
      return `<div class="day">
        <div class="day-head"><h3>${day[0]} · ${day[1]}</h3><span>${dayActivities.length} atividades</span></div>
        <div class="timeline">
          ${dayActivities.map(a=>`<div class="activity"><div class="activity-time">${a[0]}</div><div class="activity-body"><strong>${a[1]}</strong><span>${a[2]}</span></div><span class="tag">${a[3]}</span></div>`).join("")}
        </div>
      </div>`;
    }).join("");
  }

  function selectOption(button, type){
    const parent = button.closest('.options');
    parent.querySelectorAll('.select-btn').forEach(b=>{
      b.classList.remove('selected');
      b.textContent='Escolher';
    });
    button.classList.add('selected');
    button.textContent='Selecionado';

    const option = button.closest('.option');
    const priceText = option.querySelector('.price').textContent.replace('€','').trim();
    if(type === 'transport') selectedTransport = Number(priceText);
    if(type === 'hotel') selectedHotel = Number(priceText);
    updateTotals();
    showToast(type === 'transport' ? 'Transporte atualizado.' : 'Alojamento atualizado.');
  }

  function updateTotals(){
    document.getElementById('transport-total').textContent = `€${selectedTransport}`;
    document.getElementById('hotel-total').textContent = `€${selectedHotel}`;
    document.getElementById('grand-total').textContent = `€${selectedTransport + selectedHotel}`;
  }

  function generateTrip(){
    const origin = document.getElementById('origin').value || 'Origem';
    const destination = document.getElementById('destination').value || 'Destino';
    const start = document.getElementById('start').value;
    const end = document.getElementById('end').value;
    if(start && end && new Date(end) < new Date(start)){
      showToast('A data de regresso deve ser posterior à ida.');
      return;
    }
    const nights = start && end ? Math.max(1, Math.round((new Date(end)-new Date(start))/86400000)) : 4;
    document.getElementById('duration-total').textContent = `${nights} noites`;
    document.querySelector('.route-title').textContent = `${origin} → ${destination}`;
    renderItinerary();
    updateTotals();
    showToast('Itinerário personalizado gerado.');
    scrollToId('itinerario');
  }

  function saveTrip(){
    const trip = {
      origem: document.getElementById('origin').value,
      destino: document.getElementById('destination').value,
      ida: document.getElementById('start').value,
      regresso: document.getElementById('end').value,
      transportes: selectedTransport,
      alojamento: selectedHotel
    };
    localStorage.setItem('tripflow-trip', JSON.stringify(trip));
    showToast('Viagem guardada neste dispositivo.');
  }

  function scrollToId(id){
    document.getElementById(id).scrollIntoView({behavior:'smooth',block:'start'});
  }

  function showToast(message){
    const toast = document.getElementById('toast');
    toast.textContent = message;
    toast.classList.add('show');
    clearTimeout(window.toastTimer);
    window.toastTimer = setTimeout(()=>toast.classList.remove('show'),2600);
  }

  renderItinerary();
  updateTotals();
</script>
